// ── 3. STRUCT GLOBAL ─────────────────────────────────────────
// FIX: Los campos de struct terminan con ; (no con NL solos).
// La gramática define: campo_struct : tipo ID terminador
// Un NL cuenta como terminador, pero el problema es que
// la llave de apertura { va seguida de NL y el parser
// necesita ver campos inmediatamente. Usa ; para mayor seguridad.
struct Persona {
    string Nombre;
    int Edad;
    bool EsEstudiante;
}

struct Producto {
    int id;
    string nombre;
    float64 precio;
}

struct Nodo {
    int valor;
}


// ── 4. FUNCIONES GLOBALES ────────────────────────────────────

func saludar() {
    fmt.Println("Hola desde una funcion!")
}

func sumar(a int, b int) int {
    return a + b
}

func dividir(a float64, b float64) float64 {
    return a / b
}

func esMayor(a int, b int) bool {
    return a > b
}

func concatenar(a string, b string) string {
    return a + b
}

func agregarElemento(s []int, val int) []int {
    return append(s, val)
}

func actualizarNombre(p Persona, nuevoNombre string) {
    p.Nombre = nuevoNombre
}

func factorial(n int) int {
    if n <= 1 {
        return 1
    }
    return n * factorial(n - 1)
}


// ── 5. FUNCIÓN MAIN ──────────────────────────────────────────
func main() {

    // ── 5.1 Declaración de variables ─────────────────────────
    x := 10
    y := 20
    nombre := "GoScript"
    activo := false
    caracter := 'Z'

    var a int = 5
    var b float64 = 2.5
    var c string = "prueba"
    var d bool = true
    var e rune = 'X'
    var f int

    fmt.Println("Variables:", x, y, nombre, activo, caracter)
    fmt.Println("Var con tipo:", a, b, c, d, e, f)


    // ── 5.2 Operaciones aritméticas ──────────────────────────
    suma := x + y
    resta := y - x
    producto := x * a
    cociente := y / x
    modulo := y % 3
    negacion := -x
    var mixto float64 = 5
    mezcla := x + b

    fmt.Println("Suma:", suma)
    fmt.Println("Resta:", resta)
    fmt.Println("Producto:", producto)
    fmt.Println("Cociente:", cociente)
    fmt.Println("Modulo:", modulo)
    fmt.Println("Negacion:", negacion)
    fmt.Println("Mixto float:", mixto)
    fmt.Println("Mezcla int+float:", mezcla)

    // Operaciones de suma con distintos tipos (tabla sección 4.3.1)
    strInt := "manzanas" + 5
    strFloat := "precio: " + 1.5
    intBool := 5 + true
    intRune := 3 + 'A'
    repetido := 3 * "ha"
    fmt.Println("string+int:", strInt)
    fmt.Println("string+float:", strFloat)
    fmt.Println("int+bool:", intBool)
    fmt.Println("int+rune:", intRune)
    fmt.Println("int*string:", repetido)


    // ── 5.3 Operadores de asignación compuesta ───────────────
    // FIX: Las asignaciones compuestas como sentencia sola
    // necesitan estar en su propia línea con terminador.
    // El parser las trata como `asignacion` dentro de `sentencia`.
    x += 5;
    fmt.Println("x += 5:", x)
    x -= 3;
    fmt.Println("x -= 3:", x)
    x *= 2;
    fmt.Println("x *= 2:", x)
    x /= 4;
    fmt.Println("x /= 4:", x)
    x %= 3;
    fmt.Println("x %= 3:", x)
    nombre += " es genial";
    fmt.Println("nombre +=:", nombre)


    // ── 5.4 Operadores relacionales y lógicos ────────────────
    r1 := 10 == 10
    r2 := 10 != 5
    r3 := 3 < 5
    r4 := 5 > 3
    r5 := 3 <= 3
    r6 := 4 >= 4
    r7 := "abc" == "abc"
    r8 := 'a' < 'b'
    fmt.Println("Relacionales:", r1, r2, r3, r4, r5, r6, r7, r8)

    l1 := true && false
    l2 := true || false
    l3 := !true
    fmt.Println("Logicos:", l1, l2, l3)


    // ── 5.5 Signos de agrupación ─────────────────────────────
    agrupado := 3 - (1 + 3) * 32 / 90
    complejo := (10 + 2) * (3 - 1) / (4 % 3)
    fmt.Println("Agrupado:", agrupado)
    fmt.Println("Complejo:", complejo)


    // ── 5.6 Sentencia IF / ELSE IF / ELSE ────────────────────
    edad := 18

    if edad >= 18 {
        fmt.Println("Es mayor de edad")
    }

    if edad > 20 {
        fmt.Println("Mayor de 20")
    } else if edad == 18 {
        fmt.Println("Tiene exactamente 18")
    } else {
        fmt.Println("Menor de 18")
    }

    if (edad >= 18) {
        fmt.Println("Mayor de edad con parentesis")
    }

    if edad >= 18 {
        if edad < 65 {
            fmt.Println("Adulto en edad laboral")
        }
    }


    // ── 5.7 Sentencia SWITCH / CASE ──────────────────────────
    dia := 3
    switch dia {
    case 1:
        fmt.Println("Lunes")
    case 2:
        fmt.Println("Martes")
    case 3:
        fmt.Println("Miercoles")
    case 4:
        fmt.Println("Jueves")
    case 5:
        fmt.Println("Viernes")
    default:
        fmt.Println("Fin de semana")
    }

    color := "rojo"
    switch color {
    case "rojo":
        fmt.Println("Color: Rojo")
    case "azul":
        fmt.Println("Color: Azul")
    default:
        fmt.Println("Color desconocido")
    }

    numero := 7
    switch numero {
    case 1:
        fmt.Println("Uno")
    case 7:
        fmt.Println("Siete")
    }


    // ── 5.8 FOR estilo while ─────────────────────────────────
    i := 1
    for i <= 5 {
        fmt.Println("i:", i)
        i++
    }


    // ── 5.9 FOR clásico ──────────────────────────────────────
    for j := 0; j < 5; j++ {
        fmt.Println("j:", j)
    }

    for k := 10; k > 0; k-- {
        fmt.Println("k:", k)
    }

    for m := 0; m < 20; m += 5 {
        fmt.Println("m:", m)
    }


    // ── 5.10 BREAK y CONTINUE ────────────────────────────────
    for n := 0; n < 10; n++ {
        if n == 5 {
            fmt.Println("break en n=5")
            break
        }
        fmt.Println("n:", n)
    }

    for p := 1; p <= 10; p++ {
        if p % 2 == 0 {
            continue
        }
        fmt.Println("impar:", p)
    }


    // ── 5.11 SLICES ──────────────────────────────────────────
    numeros := []int{10, 20, 30, 40, 50}
    palabras := []string{"hola", "mundo", "go"}
    booleanos := []bool{true, false, true}
    flotantes := []float64{1.1, 2.2, 3.3}
    var sliceVacio []int

    fmt.Println("Slice int:", numeros)
    fmt.Println("Slice string:", palabras)
    fmt.Println("Slice bool:", booleanos)
    fmt.Println("Slice float64:", flotantes)
    fmt.Println("Slice vacio:", sliceVacio)

    // Acceso a elemento
    fmt.Println("numeros[0]:", numeros[0])
    fmt.Println("numeros[2]:", numeros[2])
    fmt.Println("palabras[1]:", palabras[1])

    // FIX: Asignación a índice — usar ; para forzar terminador
    // El parser necesita saber que `numeros[2]` es el LHS de
    // una asignación, no el inicio de una expresión.
    numeros[2] = 999;
    fmt.Println("numeros[2] modificado:", numeros[2])
    fmt.Println("Slice tras modificacion:", numeros)

    numeros[0] += 100;
    fmt.Println("numeros[0] += 100:", numeros[0])

    // len y append
    longitud := len(numeros)
    fmt.Println("len(numeros):", longitud)

    numeros = append(numeros, 60);
    fmt.Println("append 60:", numeros)
    numeros = append(numeros, 70);
    fmt.Println("append 70:", numeros)

    // slices.Index y strings.Join
    idx := slices.Index(numeros, 999)
    fmt.Println("slices.Index(999):", idx)
    idx2 := slices.Index(numeros, 9999)
    fmt.Println("slices.Index(9999):", idx2)

    frase := strings.Join(palabras, " ")
    fmt.Println("strings.Join:", frase)
    fraseComa := strings.Join(palabras, ", ")
    fmt.Println("strings.Join coma:", fraseComa)


    // ── 5.12 FOR RANGE ───────────────────────────────────────
    for idx, val := range numeros {
        fmt.Println("range idx:", idx, "val:", val)
    }

    for val := range palabras {
        fmt.Println("range val:", val)
    }


    // ── 5.13 SLICES MULTIDIMENSIONALES ───────────────────────
    // FIX: El literal [][]int{...} con filas en líneas separadas
    // genera NL dentro del literal. La gramática procesa
    // lista_expresiones separadas por COMMA. Para evitar que
    // el NL entre filas corte el literal, pon cada fila en la
    // misma línea o usa ; al final de cada fila interna.
    matriz := [][]int{{1, 2, 3}, {4, 5, 6}, {7, 8, 9}}

    fmt.Println("matriz[0][0]:", matriz[0][0])
    fmt.Println("matriz[1][2]:", matriz[1][2])
    fmt.Println("matriz[2][1]:", matriz[2][1])

    matriz[0][0] = 100;
    fmt.Println("matriz[0][0] modificado:", matriz[0][0])

    jagged := [][]int{{1, 2, 3}, {4, 5}, {6, 7, 8, 9}}
    fmt.Println("jagged[2][3]:", jagged[2][3])

    nuevaFila := []int{10, 11, 12}
    matriz = append(matriz, nuevaFila);
    fmt.Println("nueva fila matriz[3][1]:", matriz[3][1])

    for fila, rowVal := range matriz {
        fmt.Println("fila:", fila, "contenido:", rowVal)
    }


    // ── 5.14 STRUCTS ─────────────────────────────────────────
    // FIX: La instanciación usa la forma: TipoStruct varNombre = { ... }
    // que en la gramática es: ID ID ASSIGN LBRACE lista_campos_instancia RBRACE
    Persona p1 = { Nombre: "Alice", Edad: 25, EsEstudiante: true }
    Persona p2 = { Nombre: "Bob", Edad: 30, EsEstudiante: false }
    Producto prod1 = { id: 1, nombre: "Laptop", precio: 999.99 }

    fmt.Println("Struct p1:", p1)
    fmt.Println("Struct prod1:", prod1)

    // Acceso a atributos
    fmt.Println("Nombre p1:", p1.Nombre)
    fmt.Println("Edad p1:", p1.Edad)
    fmt.Println("EsEstudiante p1:", p1.EsEstudiante)
    fmt.Println("precio prod1:", prod1.precio)

    // FIX: Asignación a atributo — usar ; para forzar terminador
    p1.Nombre = "Alicia";
    p1.Edad = 26;
    fmt.Println("Nombre modificado:", p1.Nombre)
    fmt.Println("Edad modificada:", p1.Edad)

    prod1.precio += 50.0;
    fmt.Println("precio += 50:", prod1.precio)

    // Llamada a función con struct
    actualizarNombre(p2, "Roberto");
    fmt.Println("Nombre p2 tras funcion:", p2.Nombre)


    // ── 5.15 FUNCIONES EMBEBIDAS ─────────────────────────────
    numStr := strconv.Atoi("123")
    fmt.Println("strconv.Atoi:", numStr)

    floatStr := strconv.ParseFloat("3.14159")
    fmt.Println("strconv.ParseFloat:", floatStr)

    tipoInt := reflect.TypeOf(x)
    tipoFloat := reflect.TypeOf(b)
    tipoString := reflect.TypeOf(nombre)
    tipoBool := reflect.TypeOf(activo)
    tipoSlice := reflect.TypeOf(numeros)
    tipoStruct := reflect.TypeOf(p1)
    fmt.Println("tipo int:", tipoInt)
    fmt.Println("tipo float64:", tipoFloat)
    fmt.Println("tipo string:", tipoString)
    fmt.Println("tipo bool:", tipoBool)
    fmt.Println("tipo slice:", tipoSlice)
    fmt.Println("tipo struct:", tipoStruct)


    // ── 5.16 LLAMADAS A FUNCIONES ────────────────────────────
    // FIX: Una llamada a función como sentencia sola (sin := ni =)
    // se parsea como `expresion`. Necesita terminador explícito
    // cuando la siguiente línea también empieza con expresión.
    saludar();
    resultSuma := sumar(3, 7)
    fmt.Println("sumar(3,7):", resultSuma)

    resultDiv := dividir(10.0, 3.0)
    fmt.Println("dividir(10,3):", resultDiv)

    resultMayor := esMayor(10, 5)
    fmt.Println("esMayor(10,5):", resultMayor)

    resultConcat := concatenar("Hola ", "Mundo")
    fmt.Println("concatenar:", resultConcat)

    listaInt := []int{1, 2, 3}
    listaInt = agregarElemento(listaInt, 4);
    fmt.Println("agregarElemento:", listaInt)

    fact5 := factorial(5)
    fmt.Println("factorial(5):", fact5)


    // ── 5.17 BLOQUES INDEPENDIENTES ──────────────────────────
    z := 0
    fmt.Println("z antes del bloque:", z)
    {
        j2 := 99
        z2 := 40
        fmt.Println("j2 en bloque:", j2)
        fmt.Println("z2 en bloque shadow:", z2)
    }
    fmt.Println("z fuera del bloque:", z)


    // ── 5.18 VARIABLES GLOBALES ──────────────────────────────
    fmt.Println("globalInt:", globalInt)
    fmt.Println("globalFloat:", globalFloat)
    fmt.Println("globalString:", globalString)
    fmt.Println("PI:", PI)
    fmt.Println("SALUDO:", SALUDO)


    // ── 5.19 fmt.Print (sin salto de línea) ──────────────────
    fmt.Print("Sin salto ")
    fmt.Print("de linea ")
    fmt.Println("fin")


    // ── 5.20 EXPRESIONES COMPLEJAS ───────────────────────────
    compleja1 := (10 + 2) * factorial(3) - sumar(1, 1)
    fmt.Println("expresion compleja 1:", compleja1)

    sliceAux := []int{5, 10, 15}
    compleja2 := sliceAux[0] + sliceAux[1] * 2
    fmt.Println("expresion compleja 2:", compleja2)

    edadDoble := p1.Edad * 2
    fmt.Println("edad doble:", edadDoble)

    if len(numeros) > 5 && p1.Edad >= 18 {
        fmt.Println("slice largo y persona adulta")
    }

    switch sumar(2, 3) {
    case 5:
        fmt.Println("2+3 = 5, correcto")
    default:
        fmt.Println("resultado inesperado")
    }

    fmt.Println("=== FIN DE PRUEBAS ===")
}
