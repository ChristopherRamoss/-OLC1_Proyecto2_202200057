// Lenguaje/Interprete.js
const { agregarError } = require('../Util/Errores');
const { listaSimbolos, Simbolo } = require('../Util/TablaSimbolos');
const Entorno = require('../Util/Entorno');

/* ── Señales de control ───────────────────────────────── */
class SeñalBreak    { constructor() { this.esBreak    = true; } }
class SeñalContinue { constructor() { this.esContinue = true; } }
class SeñalReturn   { constructor(v) { this.esReturn  = true; this.valor = v; } }

/* ── Registro global de structs y funciones ───────────── */
const structs   = {};
const funciones = {};

/* ── Salida acumulada de fmt.Println / fmt.Print ─────── */
let salida = [];

/* ============================================================
   HELPERS
   ============================================================ */
const registrar = (id, tipoSimbolo, tipoDato, ambito) => {
    listaSimbolos.push(new Simbolo(id, tipoSimbolo, tipoDato, ambito, 0, 0));
};

const valorRune = (r) => {
    if (typeof r === 'number') return r;
    if (typeof r === 'string' && r.startsWith("'") && r.endsWith("'")) {
        const inner = r.slice(1, -1);
        if (inner === '\\n') return 10;
        if (inner === '\\t') return 9;
        if (inner === '\\r') return 13;
        if (inner.startsWith('\\')) return inner.charCodeAt(1);
        return inner.charCodeAt(0);
    }
    return 0;
};

const esRune = (v) =>
    typeof v === 'string' && v.startsWith("'") && v.endsWith("'");

const resolverRune = (v) => esRune(v) ? valorRune(v) : v;

const tipoDe = (v) => {
    if (v === null || v === undefined) return 'nil';
    if (typeof v === 'boolean') return 'bool';
    if (Number.isInteger(v)) return 'int';
    if (typeof v === 'number') return 'float64';
    if (typeof v === 'string') return 'string';
    if (Array.isArray(v)) return 'slice';
    if (typeof v === 'object' && v.__struct) return v.__struct;
    return 'desconocido';
};

const valorDefault = (tipo) => {
    if (!tipo) return null;
    if (tipo === 'int' || tipo === 'rune') return 0;
    if (tipo === 'float64') return 0.0;
    if (tipo === 'string') return '';
    if (tipo === 'bool') return false;
    if (tipo.startsWith('[]')) return [];
    if (structs[tipo]) {
        const inst = { __struct: tipo };
        (structs[tipo].campos || []).forEach(c => { inst[c.nombre] = valorDefault(c.tipoDato); });
        return inst;
    }
    return null;
};

const formatear = (v) => {
    if (v === null || v === undefined) return 'nil';
    if (typeof v === 'boolean') return v ? 'true' : 'false';
    if (typeof v === 'number') return String(v);
    if (typeof v === 'string') return v;
    if (Array.isArray(v)) {
        if (v.length > 0 && Array.isArray(v[0]))
            return '[' + v.map(r => '[' + r.map(formatear).join(' ') + ']').join(' ') + ']';
        return '[' + v.map(formatear).join(' ') + ']';
    }
    if (typeof v === 'object' && v.__struct) {
        const campos = Object.entries(v)
            .filter(([k]) => k !== '__struct')
            .map(([k, val]) => `${k}: ${formatear(val)}`)
            .join(', ');
        return `${v.__struct}{${campos}}`;
    }
    return String(v);
};

/* ============================================================
   OPERACIONES ARITMÉTICAS
   ============================================================ */
const suma = (a, b) => {
    a = resolverRune(a); b = resolverRune(b);
    const ta = tipoDe(a), tb = tipoDe(b);
    if (ta === 'string' || tb === 'string') return String(a) + String(b);
    if (ta === 'bool' && tb === 'bool') return a || b;
    return a + b;
};
const resta = (a, b) => {
    a = resolverRune(a); b = resolverRune(b);
    const ta = tipoDe(a), tb = tipoDe(b);
    if (ta === 'bool' && tb === 'bool') return a && !b;
    const na = ta === 'bool' ? (a ? 1 : 0) : a;
    const nb = tb === 'bool' ? (b ? 1 : 0) : b;
    return na - nb;
};
const mult = (a, b) => {
    a = resolverRune(a); b = resolverRune(b);
    const ta = tipoDe(a), tb = tipoDe(b);
    if (ta === 'int'    && tb === 'string') return b.repeat(a);
    if (ta === 'string' && tb === 'int')   return a.repeat(b);
    if (ta === 'bool'   && tb === 'bool')  return a && b;
    const na = ta === 'bool' ? (a ? 1 : 0) : a;
    const nb = tb === 'bool' ? (b ? 1 : 0) : b;
    return na * nb;
};
const divi = (a, b) => {
    a = resolverRune(a); b = resolverRune(b);
    if (b === 0) { agregarError('Semántico', 'División por cero', 0, 0); return 0; }
    const r = a / b;
    return (Number.isInteger(a) && Number.isInteger(b)) ? Math.trunc(r) : r;
};
const modu = (a, b) => {
    a = resolverRune(a); b = resolverRune(b);
    if (b === 0) { agregarError('Semántico', 'Módulo por cero', 0, 0); return 0; }
    return Math.trunc(a) % Math.trunc(b);
};
const opAsig = (op, act, val) => {
    switch (op) {
        case '+=': return suma(act, val);
        case '-=': return resta(act, val);
        case '*=': return mult(act, val);
        case '/=': return divi(act, val);
        case '%=': return modu(act, val);
        default:   return val;
    }
};

/* ============================================================
   RECOLECTOR DE DECLARACIONES GLOBALES
   ============================================================ */
const recolectar = (ast) => {
    for (const nodo of ast) {
        if (!nodo) continue;
        if (nodo.tipo === 'decl_struct') {
            structs[nodo.nombre] = { campos: nodo.campos || [] };
            registrar(nodo.nombre, 'Struct', nodo.nombre, 'Global');
        }
        if (nodo.tipo === 'decl_func') {
            funciones[nodo.nombre] = nodo;
            registrar(nodo.nombre, 'Función', nodo.retorno || 'void', 'Global');
        }
        if (nodo.tipo === 'func_main') {
            funciones['main'] = nodo;
        }
    }
};

/* ============================================================
   EVALUADOR DE EXPRESIONES
   ============================================================ */
const evalExpr = (nodo, env) => {
    if (!nodo) return null;

    switch (nodo.tipo) {
        case 'lit_int':   return nodo.valor;
        case 'lit_float': return nodo.valor;
        case 'lit_bool':  return nodo.valor;
        case 'lit_rune':  return valorRune(nodo.valor);
        case 'nil':       return null;

        case 'lit_string': {
            let s = nodo.valor;
            if (s.startsWith('"') && s.endsWith('"')) s = s.slice(1, -1);
            s = s.replace(/\\n/g, '\n').replace(/\\t/g, '\t')
                 .replace(/\\r/g, '\r').replace(/\\"/g, '"').replace(/\\\\/g, '\\');
            return s;
        }

        case 'id': {
            const reg = env.obtener(nodo.nombre);
            if (reg === null || reg === undefined) {
                agregarError('Semántico', `Variable '${nodo.nombre}' no declarada`, 0, 0);
                return null;
            }
            return reg.valor;
        }

        case 'op_bin': {
            const izq = evalExpr(nodo.izq, env);
            const der = evalExpr(nodo.der, env);
            switch (nodo.op) {
                case '+':  return suma(izq, der);
                case '-':  return resta(izq, der);
                case '*':  return mult(izq, der);
                case '/':  return divi(izq, der);
                case '%':  return modu(izq, der);
                case '==': return izq === der;
                case '!=': return izq !== der;
                case '<':  return izq < der;
                case '>':  return izq > der;
                case '<=': return izq <= der;
                case '>=': return izq >= der;
                case '&&': return Boolean(izq) && Boolean(der);
                case '||': return Boolean(izq) || Boolean(der);
                default:   return null;
            }
        }

        case 'op_unario': {
            const v = evalExpr(nodo.operando, env);
            if (nodo.op === '-') return -v;
            if (nodo.op === '!') return !v;
            return null;
        }

        case 'inc': {
            const reg = env.obtener(nodo.nombre);
            const nuevo = (reg ? reg.valor : 0) + 1;
            env.guardar(nodo.nombre, nuevo, reg ? reg.tipo : 'int');
            return nuevo;
        }
        case 'dec': {
            const reg = env.obtener(nodo.nombre);
            const nuevo = (reg ? reg.valor : 0) - 1;
            env.guardar(nodo.nombre, nuevo, reg ? reg.tipo : 'int');
            return nuevo;
        }

        case 'acceso_indice': {
            const obj = evalExpr(nodo.objeto, env);
            const idx = evalExpr(nodo.indice, env);
            if (!Array.isArray(obj)) {
                agregarError('Semántico', 'Acceso a índice sobre valor que no es slice', 0, 0);
                return null;
            }
            if (idx < 0 || idx >= obj.length) {
                agregarError('Semántico', `Índice ${idx} fuera de rango (len=${obj.length})`, 0, 0);
                return null;
            }
            return obj[idx];
        }

        case 'acceso_atributo': {
            const obj = evalExpr(nodo.objeto, env);
            if (obj === null || obj === undefined || typeof obj !== 'object' || Array.isArray(obj)) {
                agregarError('Semántico', `Acceso a atributo '${nodo.campo}' sobre valor inválido`, 0, 0);
                return null;
            }
            return obj[nodo.campo] !== undefined ? obj[nodo.campo] : null;
        }

        case 'lit_slice': {
            const elementos = nodo.elementos || [];
            return elementos.map(e => {
                if (e && e.tipo === 'fila_slice')
                    return (e.elementos || []).map(ie => evalExpr(ie, env));
                return evalExpr(e, env);
            });
        }

        case 'lit_slice_2d': {
            const filas = nodo.filas || [];
            return filas.map(fila => (fila || []).map(e => evalExpr(e, env)));
        }

        case 'llamada': return evalLlamada(nodo, env);

        default: return null;
    }
};

/* ============================================================
   EVALUADOR DE LLAMADAS
   ============================================================ */
const evalLlamada = (nodo, env) => {
    const nombre = nodo.nombre;
    const args   = (nodo.args || []).map(a => evalExpr(a, env));

    switch (nombre) {
        case 'fmt.Println': {
            const linea = args.length === 0 ? '' : args.map(formatear).join(' ');
            salida.push(linea);
            return null;
        }
        case 'fmt.Print': {
            const texto = args.map(formatear).join(' ');
            if (salida.length > 0) {
                salida[salida.length - 1] += texto;
            } else {
                salida.push(texto);
            }
            return null;
        }
        case 'len': {
            const s = args[0];
            return Array.isArray(s) ? s.length : (typeof s === 'string' ? s.length : 0);
        }
        case 'append': {
            const arr = args[0];
            if (!Array.isArray(arr)) return [];
            return [...arr, args[1]];
        }
        case 'slices.Index': {
            const arr = args[0], buscado = args[1];
            if (!Array.isArray(arr)) return -1;
            return arr.findIndex(e => e === buscado);
        }
        case 'strings.Join': {
            const arr = args[0], sep = args[1];
            return Array.isArray(arr) ? arr.join(sep) : '';
        }
        case 'strconv.Atoi': {
            const n = parseInt(args[0], 10);
            if (isNaN(n)) { agregarError('Semántico', `strconv.Atoi: '${args[0]}' no es entero`, 0, 0); return 0; }
            return n;
        }
        case 'strconv.ParseFloat': {
            const n = parseFloat(args[0]);
            if (isNaN(n)) { agregarError('Semántico', `strconv.ParseFloat: '${args[0]}' inválido`, 0, 0); return 0.0; }
            return n;
        }
        case 'reflect.TypeOf': {
            const v = args[0];
            if (v === null || v === undefined) return 'nil';
            if (typeof v === 'boolean') return 'bool';
            if (Number.isInteger(v)) return 'int';
            if (typeof v === 'number') return 'float64';
            if (typeof v === 'string') return 'string';
            if (Array.isArray(v)) return v.length > 0 && Array.isArray(v[0]) ? '[][]int' : '[]int';
            if (typeof v === 'object' && v.__struct) return v.__struct;
            return 'desconocido';
        }
    }

    // Función de usuario
    const decl = funciones[nombre];
    if (!decl) {
        agregarError('Semántico', `Función '${nombre}' no declarada`, 0, 0);
        return null;
    }

    const envFun = new Entorno(null);
    (decl.params || []).forEach((param, i) => {
        const val = args[i] !== undefined ? args[i] : valorDefault(param.tipoDato);
        envFun.guardar(param.nombre, val, param.tipoDato);
        registrar(param.nombre, 'Parámetro', param.tipoDato, nombre);
    });

    const res = ejecutarBloque(decl.cuerpo, envFun);
    return res instanceof SeñalReturn ? res.valor : null;
};

/* ============================================================
   INSTANCIAR STRUCT
   ============================================================ */
const instanciarStruct = (tipoStruct, camposLiteral, env) => {
    if (!structs[tipoStruct]) {
        agregarError('Semántico', `Struct '${tipoStruct}' no declarado`, 0, 0);
        return null;
    }
    const inst = { __struct: tipoStruct };
    (structs[tipoStruct].campos || []).forEach(c => { inst[c.nombre] = valorDefault(c.tipoDato); });
    if (Array.isArray(camposLiteral)) {
        camposLiteral.forEach(ci => { inst[ci.campo] = evalExpr(ci.valor, env); });
    }
    return inst;
};

/* ============================================================
   ASIGNAR A ÍNDICE
   ============================================================ */
const asignarIndice = (nodo, valor, env) => {
    const idx = evalExpr(nodo.indice, env);

    if (nodo.objeto.tipo === 'id') {
        const reg = env.obtener(nodo.objeto.nombre);
        if (!reg || !Array.isArray(reg.valor)) { agregarError('Semántico', 'No es slice', 0, 0); return; }
        reg.valor[idx] = valor;
        return;
    }
    const arr = evalExpr(nodo.objeto, env);
    if (Array.isArray(arr)) arr[idx] = valor;
};

/* ============================================================
   ASIGNAR A ATRIBUTO DE STRUCT
   ============================================================ */
const asignarAtributo = (nodo, valor, env) => {
    if (nodo.objeto.tipo === 'id') {
        const reg = env.obtener(nodo.objeto.nombre);
        if (!reg || typeof reg.valor !== 'object' || reg.valor === null) {
            agregarError('Semántico', 'No se puede asignar atributo a no-struct', 0, 0);
            return;
        }
        reg.valor[nodo.campo] = valor;
        return;
    }
    const obj = evalExpr(nodo.objeto, env);
    if (obj && typeof obj === 'object') obj[nodo.campo] = valor;
};

/* ============================================================
   EJECUTOR DE BLOQUE Y LISTA
   ============================================================ */
const ejecutarBloque = (bloque, envPadre) => {
    const env = new Entorno(envPadre);
    return ejecutarLista(bloque.cuerpo, env);
};

const ejecutarLista = (lista, env) => {
    if (!Array.isArray(lista)) return null;
    for (const sent of lista) {
        if (!sent) continue;
        const res = ejecutarSentencia(sent, env);
        if (res instanceof SeñalBreak || res instanceof SeñalContinue || res instanceof SeñalReturn)
            return res;
    }
    return null;
};

/* ============================================================
   EJECUTOR DE SENTENCIA
   ============================================================ */
const ejecutarSentencia = (nodo, env) => {
    if (!nodo) return null;

    switch (nodo.tipo) {

        case 'bloque': return ejecutarBloque(nodo, env);

        case 'decl_var': {
            if (nodo.forma === 'struct_literal') {
                const inst = instanciarStruct(nodo.tipoStruct, nodo.campos, env);
                env.guardar(nodo.nombre, inst, nodo.tipoStruct);
                registrar(nodo.nombre, 'Variable', nodo.tipoStruct, 'Local');
                return null;
            }
            const val = (nodo.valor !== null && nodo.valor !== undefined)
                ? evalExpr(nodo.valor, env)
                : valorDefault(nodo.tipoDato);
            env.guardar(nodo.nombre, val, nodo.tipoDato || tipoDe(val));
            registrar(nodo.nombre, 'Variable', nodo.tipoDato || tipoDe(val), 'Local');
            return null;
        }

        case 'decl_const': {
            const val = evalExpr(nodo.valor, env);
            env.guardar(nodo.nombre, val, nodo.tipoDato || tipoDe(val));
            registrar(nodo.nombre, 'Constante', nodo.tipoDato || tipoDe(val), 'Global');
            return null;
        }

        case 'asignacion': {
            const val = evalExpr(nodo.valor, env);
            const reg  = env.obtener(nodo.nombre);
            env.guardar(nodo.nombre, val, reg ? reg.tipo : tipoDe(val));
            return null;
        }

        case 'asignacion_op': {
            const reg  = env.obtener(nodo.nombre);
            const act  = reg ? reg.valor : 0;
            const nuevo = opAsig(nodo.op, act, evalExpr(nodo.valor, env));
            env.guardar(nodo.nombre, nuevo, reg ? reg.tipo : tipoDe(nuevo));
            return null;
        }

        case 'asignacion_indice':
            asignarIndice(nodo.acceso, evalExpr(nodo.valor, env), env);
            return null;

        case 'asignacion_op_indice': {
            const act = evalExpr(nodo.acceso, env);
            asignarIndice(nodo.acceso, opAsig(nodo.op, act, evalExpr(nodo.valor, env)), env);
            return null;
        }

        case 'asignacion_atributo':
            asignarAtributo(nodo.acceso, evalExpr(nodo.valor, env), env);
            return null;

        case 'asignacion_op_atributo': {
            const act = evalExpr(nodo.acceso, env);
            asignarAtributo(nodo.acceso, opAsig(nodo.op, act, evalExpr(nodo.valor, env)), env);
            return null;
        }

        case 'if': {
            const cond = evalExpr(nodo.condicion, env);
            if (cond) return ejecutarBloque(nodo.entonces, env);
            if (nodo.sino) {
                if (nodo.sino.tipo === 'bloque') return ejecutarBloque(nodo.sino, env);
                if (nodo.sino.tipo === 'if')     return ejecutarSentencia(nodo.sino, env);
            }
            return null;
        }

        case 'for_while': {
            while (true) {
                if (!evalExpr(nodo.condicion, env)) break;
                const res = ejecutarBloque(nodo.cuerpo, env);
                if (res instanceof SeñalBreak)  break;
                if (res instanceof SeñalReturn) return res;
            }
            return null;
        }

        case 'for_clasico': {
            const envFor = new Entorno(env);
            if (nodo.init) ejecutarSentencia(nodo.init, envFor);
            while (true) {
                if (!evalExpr(nodo.condicion, envFor)) break;
                const res = ejecutarBloque(nodo.cuerpo, envFor);
                if (res instanceof SeñalBreak)  break;
                if (res instanceof SeñalReturn) return res;
                if (nodo.post) ejecutarSentencia(nodo.post, envFor);
            }
            return null;
        }

        case 'for_range': {
            const iterable = evalExpr(nodo.iterable, env);
            if (!Array.isArray(iterable)) {
                agregarError('Semántico', 'for range requiere un slice', 0, 0);
                return null;
            }
            for (let i = 0; i < iterable.length; i++) {
                const envR = new Entorno(env);
                if (nodo.indice) envR.guardar(nodo.indice, i, 'int');
                envR.guardar(nodo.valor, iterable[i], tipoDe(iterable[i]));
                const res = ejecutarBloque(nodo.cuerpo, envR);
                if (res instanceof SeñalBreak)  break;
                if (res instanceof SeñalReturn) return res;
            }
            return null;
        }

        case 'switch': {
            const expr = evalExpr(nodo.expresion, env);
            let defCase = null;
            for (const caso of (nodo.casos || [])) {
                if (caso.tipo === 'default') { defCase = caso; continue; }
                if (expr === evalExpr(caso.valor, env)) {
                    const res = ejecutarLista(caso.cuerpo, new Entorno(env));
                    if (res instanceof SeñalReturn) return res;
                    return null;
                }
            }
            if (defCase) {
                const res = ejecutarLista(defCase.cuerpo, new Entorno(env));
                if (res instanceof SeñalReturn) return res;
            }
            return null;
        }

        case 'return':
            return new SeñalReturn(nodo.valor ? evalExpr(nodo.valor, env) : null);

        case 'break':    return new SeñalBreak();
        case 'continue': return new SeñalContinue();

        case 'inc': {
            const reg = env.obtener(nodo.nombre);
            const v   = reg ? reg.valor : 0;
            env.guardar(nodo.nombre, v + 1, reg ? reg.tipo : 'int');
            return null;
        }
        case 'dec': {
            const reg = env.obtener(nodo.nombre);
            const v   = reg ? reg.valor : 0;
            env.guardar(nodo.nombre, v - 1, reg ? reg.tipo : 'int');
            return null;
        }

        case 'llamada':
            evalLlamada(nodo, env);
            return null;

        default:
            evalExpr(nodo, env);
            return null;
    }
};

/* ============================================================
   PUNTO DE ENTRADA
   ============================================================ */
const interpretar = (ast) => {
    salida = [];
    Object.keys(structs).forEach(k => delete structs[k]);
    Object.keys(funciones).forEach(k => delete funciones[k]);

    if (!Array.isArray(ast)) return { salida: [] };

    recolectar(ast);

    const envGlobal = new Entorno(null);
    for (const nodo of ast) {
        if (!nodo) continue;
        if (nodo.tipo === 'decl_var' || nodo.tipo === 'decl_const')
            ejecutarSentencia(nodo, envGlobal);
    }

    const main = funciones['main'];
    if (!main) {
        agregarError('Semántico', 'No se encontró la función main()', 0, 0);
        return { salida };
    }

    try {
        ejecutarBloque(main.cuerpo, envGlobal);
    } catch (e) {
        agregarError('Semántico', `Error de ejecución: ${e.message}`, 0, 0);
        console.error('[Interprete]', e);
    }

    return { salida };
};

module.exports = { interpretar };
