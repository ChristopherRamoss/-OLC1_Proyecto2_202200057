func sumar(a int, b int) int {
    return a + b;
}

func promedio(a float64, b float64, c float64) float64 {
    return (a + b + c) / 3.0;
}

func esMayor(edad int) bool {
    return edad >= 18;
}

func crearMensaje(nombre string, edad int, activo bool) string {
    return "Nombre: " + nombre + ", Edad: " + edad + ", Activo: " + activo;
}

func aumentarCreditos(curso Curso, extra int) {
    curso.creditos = curso.creditos + extra;
}

func agregarNumero(lista []int, valor int) []int {
    return append(lista, valor);
}

func sumarSlice(valores []int) int {
    total := 0;
    for i := 0; i < len(valores); i++ {
        total = total + valores[i];
    }
    return total;
}


func imprimirPares(valores []int) {
    for i := 0; i < len(valores); i++ {
        if valores[i] % 2 == 0 {
            fmt.Println("Par:", valores[i]);
        } else {
            fmt.Println("Impar:", valores[i]);
        }
    }
}



func buscarEstado(nota int) string {
    switch nota {
        case 100:
            return "Excelente";
        case 80:
            return "Muy bien";
        case 60:
            return "Aprobado";
        default:
            return "Revisar";
    }
}
